### Out of scope

- 多窗口、多会话（V1.0 单窗口；⌘N 只重置当前工作区）。
- Apple Help Book 打包（V1.0 帮助为应用内窗口）。
- Sparkle 自动更新、设置云同步、设置导入导出。
- 控制台渲染实现本身（04 已完成；本文档只触发其投影重建）。
- 日志轮转实现（06 已完成；本文档只向其传递新阈值供下次开始记录使用）。
- 深浅色视觉细节打磨与 VoiceOver 审计（08）。

### 明确禁止事项

- 禁止在设置变更时调用任何"重新打开串口 / 重写当前 SerialConfiguration"的路径（§7.1：当前连接参数不被静默修改）。
- 禁止在设置广播中清空或重建 `RecordStore` 的原始数据；只允许重建展示投影。
- 禁止用 Alert 表达设置校验失败（设计规范 §8.6：输入框内校验 → 局部提示 → 状态栏 → Alert；只有中断会话/数据丢失/需要选择才弹 Alert）。
- 禁止在 Domain/Application 层 import SwiftUI / AppKit（外观应用到 `NSApp.appearance` 的代码只允许在 App/Presentation 层）。
- 禁止引入新第三方依赖（UserDefaults、Observation、SwiftUI Commands 均为系统能力）。

### 6.1 设置模型（`Domain/Settings/AppSettings.swift`）

```swift
import Foundation

/// 设置六分类。用于广播变更时告知订阅者"哪一类变了"，从而决定生效路径。
public enum SettingsCategory: String, Codable, Sendable, CaseIterable {
    case serialDefaults      // 串口默认值
    case display             // 显示与可读性
    case dataFormat          // 数据格式
    case retention           // 数据保留
    case logging             // 日志
    case sessionAndPresets   // 会话与预设
}

// MARK: - 1. 串口默认值（生效：新会话或下一次连接；不得静默修改当前连接参数）
public struct SerialDefaultsSettings: Codable, Equatable, Sendable {
    public var defaultBaudRate: Int                 // 默认 115200
    public var defaultDataBits: DataBits            // 默认 .eight
    public var defaultParity: Parity                // 默认 .none
    public var defaultStopBits: StopBits            // 默认 .one
    public var defaultFlowControl: FlowControl      // 默认 .none
    public var defaultDisplayFormat: DisplayFormat  // 默认 .ascii（新会话的接收/发送默认模式）
    public var refreshPortsOnLaunch: Bool           // 默认 true：启动时自动刷新端口列表

    public static let `default` = SerialDefaultsSettings(
        defaultBaudRate: 115200, defaultDataBits: .eight, defaultParity: .none,
        defaultStopBits: .one, defaultFlowControl: .none,
        defaultDisplayFormat: .ascii, refreshPortsOnLaunch: true)
}

// MARK: - 2. 显示与可读性（生效：即时；不丢失当前记录）
public enum AppearancePreference: String, Codable, Sendable, CaseIterable {
    case system, light, dark
}

public struct DisplaySettings: Codable, Equatable, Sendable {
    public var consoleFontName: String       // 默认 "Menlo"；要求系统等宽字体
    public var consoleFontSize: Double       // 默认 12，合法范围 10...18
    public var lineSpacing: Double           // 默认 2（pt），合法范围 0...12
    public var appearance: AppearancePreference // 默认 .system
    public var showTimestamps: Bool          // 默认 true
    public var autoScroll: Bool              // 默认 true

    public static let `default` = DisplaySettings(
        consoleFontName: "Menlo", consoleFontSize: 12, lineSpacing: 2,
        appearance: .system, showTimestamps: true, autoScroll: true)
}

// MARK: - 3. 数据格式（生效：下一次显示或发送；不重写已有原始数据）
public enum TextEncodingPreference: String, Codable, Sendable, CaseIterable {
    case utf8        // 默认
    case gb18030
    case isoLatin1
    case ascii

    public var stringEncoding: String.Encoding {
        switch self {
        case .utf8: return .utf8
        case .gb18030: return String.Encoding(rawValue: CFStringConvertEncodingToNSStringEncoding(CFStringEncoding(CFStringEncodings.GB_18030_2000.rawValue)))
        case .isoLatin1: return .isoLatin1
        case .ascii: return .ascii
        }
    }
}

public enum HexSeparator: String, Codable, Sendable, CaseIterable {
    case space   // 默认："55 AA 01"
    case none    // "55AA01"
    case comma   // "55,AA,01"
}

public enum LineEnding: String, Codable, Sendable, CaseIterable {
    case none, lf, cr, crlf   // 默认 .crlf（发送时追加）

    public var bytes: [UInt8] {
        switch self {
        case .none: return []
        case .lf: return [0x0A]
        case .cr: return [0x0D]
        case .crlf: return [0x0D, 0x0A]
        }
    }
}

public struct DataFormatSettings: Codable, Equatable, Sendable {
    public var textEncoding: TextEncodingPreference // 默认 .utf8
    public var hexSeparator: HexSeparator           // 默认 .space
    public var hexUppercase: Bool                   // 默认 true
    public var hexBytesPerLine: Int                 // 默认 0 = 不强制换行；合法范围 0...64
    public var lineEnding: LineEnding               // 默认 .crlf

    public static let `default` = DataFormatSettings(
        textEncoding: .utf8, hexSeparator: .space, hexUppercase: true,
        hexBytesPerLine: 0, lineEnding: .crlf)
}

// MARK: - 4. 数据保留（生效：即时；下调阈值后按最早优先裁剪）
public struct RetentionSettings: Codable, Equatable, Sendable {
    public var maxRecordCount: Int        // 默认 50_000；合法范围 1_000...500_000
    public var maxBufferBytes: Int        // 默认 16 MiB；合法范围 1 MiB...256 MiB
    public var showTruncationNotice: Bool // 默认 true：显示累计截断次数与最后截断时间

    public static let `default` = RetentionSettings(
        maxRecordCount: 50_000,
        maxBufferBytes: 16 * 1024 * 1024,
        showTruncationNotice: true)
}

// MARK: - 5. 日志（生效：新日志文件；进行中的日志保持当前策略）
public struct LoggingSettings: Codable, Equatable, Sendable {
    /// 用户选择的日志目录（安全作用域书签）；nil 表示使用默认目录
    public var directoryBookmark: Data?
    /// 目录的展示用路径缓存（书签解析失败时兜底显示）
    public var directoryPathDisplay: String   // 默认 "~/Documents/Serial Monitor Logs"
    /// 命名模板，支持 {date} {time} {port} 占位符
    public var fileNameTemplate: String       // 默认 "SerialMonitor_{date}_{time}_{port}"
    public var autoLoggingEnabled: Bool       // 默认 false：连接成功时自动开始记录
    public var rotationThresholdBytes: Int    // 默认 10 MiB；合法范围 1 MiB...1 GiB

    public static let `default` = LoggingSettings(
        directoryBookmark: nil,
        directoryPathDisplay: "~/Documents/Serial Monitor Logs",
        fileNameTemplate: "SerialMonitor_{date}_{time}_{port}",
        autoLoggingEnabled: false,
        rotationThresholdBytes: 10 * 1024 * 1024)
}

// MARK: - 6. 会话与预设（生效：重启后恢复；自动发送不得在启动时自动恢复运行）
public struct SessionPresetSettings: Codable, Equatable, Sendable {
    public var restoreLastConnectionParameters: Bool // 默认 true
    public var restoreAutoSendDraft: Bool            // 默认 true：仅恢复内容与间隔草稿

    public static let `default` = SessionPresetSettings(
        restoreLastConnectionParameters: true, restoreAutoSendDraft: true)
}

// MARK: - 根模型
public struct AppSettings: Codable, Equatable, Sendable {
    public var schemaVersion: Int
    public var serialDefaults: SerialDefaultsSettings
    public var display: DisplaySettings
    public var dataFormat: DataFormatSettings
    public var retention: RetentionSettings
    public var logging: LoggingSettings
    public var sessionAndPresets: SessionPresetSettings

    public static let currentSchemaVersion = 1

    public static let `default` = AppSettings(
        schemaVersion: currentSchemaVersion,
        serialDefaults: .default, display: .default, dataFormat: .default,
        retention: .default, logging: .default, sessionAndPresets: .default)
}

/// 一次设置变更的广播载荷。
public struct AppSettingsChange: Sendable {
    public let previous: AppSettings
    public let current: AppSettings
    public let changedCategories: Set<SettingsCategory>
}
```

### 6.2 校验规则（`Domain/Settings/SettingsValidation.swift`）

```swift
import Foundation

/// 设置字段标识，用于把校验错误定位到具体控件。
public enum SettingsField: String, Sendable, Hashable {
    case baudRate, fontName, fontSize, lineSpacing
    case hexBytesPerLine, maxRecordCount, maxBufferBytes
    case logDirectory, fileNameTemplate, rotationThresholdBytes
}

public struct SettingsIssue: Equatable, Sendable {
    public let field: SettingsField
    public let message: String   // 用户可读、就近展示（不进 Alert）
}

public enum SettingsValidation {
    public static let fontSizeRange: ClosedRange<Double> = 10...18
    public static let lineSpacingRange: ClosedRange<Double> = 0...12
    public static let hexBytesPerLineRange: ClosedRange<Int> = 0...64
    public static let maxRecordCountRange: ClosedRange<Int> = 1_000...500_000
    public static let maxBufferBytesRange: ClosedRange<Int> = (1 << 20)...(256 << 20)
    public static let rotationThresholdRange: ClosedRange<Int> = (1 << 20)...(1 << 30)
    public static let allowedBaudRates: [Int] = [
        300, 1200, 2400, 4800, 9600, 19200, 38400, 57600,
        115200, 230400, 460800, 921600
    ]

    /// 全量校验。返回空数组表示可保存；否则调用方必须拒绝保存。
    public static func validate(_ s: AppSettings) -> [SettingsIssue] {
        var issues: [SettingsIssue] = []
        if !allowedBaudRates.contains(s.serialDefaults.defaultBaudRate) {
            issues.append(.init(field: .baudRate, message: "不支持的波特率 \(s.serialDefaults.defaultBaudRate)"))
        }
        if !fontSizeRange.contains(s.display.consoleFontSize) {
            issues.append(.init(field: .fontSize, message: "字号须在 10–18 pt 之间"))
        }
        if !lineSpacingRange.contains(s.display.lineSpacing) {
            issues.append(.init(field: .lineSpacing, message: "行距须在 0–12 pt 之间"))
        }
        if !hexBytesPerLineRange.contains(s.dataFormat.hexBytesPerLine) {
            issues.append(.init(field: .hexBytesPerLine, message: "每行字节数须在 0–64 之间"))
        }
        if !maxRecordCountRange.contains(s.retention.maxRecordCount) {
            issues.append(.init(field: .maxRecordCount, message: "最大记录数须在 1,000–500,000 之间"))
        }
        if !maxBufferBytesRange.contains(s.retention.maxBufferBytes) {
            issues.append(.init(field: .maxBufferBytes, message: "最大字节数须在 1–256 MiB 之间"))
        }
        if !rotationThresholdRange.contains(s.logging.rotationThresholdBytes) {
            issues.append(.init(field: .rotationThresholdBytes, message: "轮转阈值须在 1 MiB–1 GiB 之间"))
        }
        if s.logging.fileNameTemplate.trimmingCharacters(in: .whitespaces).isEmpty
            || s.logging.fileNameTemplate.contains("/")
            || s.logging.fileNameTemplate.contains("..") {
            issues.append(.init(field: .fileNameTemplate, message: "命名规则不能为空，且不得包含路径分隔符"))
        }
        return issues
    }

    /// 仅用于"持久化数据损坏/越界"的加载兜底：把越界值钳制回合法区间。
    /// 用户编辑路径不得调用本方法——用户输入的无效值必须拒绝而非静默修正。
    public static func clampedForLoad(_ s: AppSettings) -> AppSettings {
        var s = s
        s.display.consoleFontSize = min(max(s.display.consoleFontSize, fontSizeRange.lowerBound), fontSizeRange.upperBound)
        s.display.lineSpacing = min(max(s.display.lineSpacing, lineSpacingRange.lowerBound), lineSpacingRange.upperBound)
        s.dataFormat.hexBytesPerLine = min(max(s.dataFormat.hexBytesPerLine, hexBytesPerLineRange.lowerBound), hexBytesPerLineRange.upperBound)
        s.retention.maxRecordCount = min(max(s.retention.maxRecordCount, maxRecordCountRange.lowerBound), maxRecordCountRange.upperBound)
        s.retention.maxBufferBytes = min(max(s.retention.maxBufferBytes, maxBufferBytesRange.lowerBound), maxBufferBytesRange.upperBound)
        s.logging.rotationThresholdBytes = min(max(s.logging.rotationThresholdBytes, rotationThresholdRange.lowerBound), rotationThresholdRange.upperBound)
        if !allowedBaudRates.contains(s.serialDefaults.defaultBaudRate) {
            s.serialDefaults.defaultBaudRate = SerialDefaultsSettings.default.defaultBaudRate
        }
        if s.logging.fileNameTemplate.isEmpty || s.logging.fileNameTemplate.contains("/") {
            s.logging.fileNameTemplate = LoggingSettings.default.fileNameTemplate
        }
        return s
    }
}
```

### 6.3 持久化（`Infrastructure/Persistence/SettingsStore.swift`）

```swift
import Foundation

public protocol SettingsStoring: Sendable {
    /// 返回 (设置, 是否发生过损坏回退)。无数据时返回 (.default, false)。
    func load() -> (settings: AppSettings, didRecoverFromCorruption: Bool)
    func save(_ settings: AppSettings) throws
}

public final class UserDefaultsSettingsStore: SettingsStoring {
    private let defaults: UserDefaults
    private static let key = "SerialMonitor.AppSettings.v1"
    private static let corruptBackupKeyPrefix = "SerialMonitor.AppSettings.v1.corrupt."

    public init(defaults: UserDefaults = .standard) { self.defaults = defaults }

    public func load() -> (AppSettings, Bool) {
        guard let data = defaults.data(forKey: Self.key) else { return (.default, false) }
        do {
            let decoded = try JSONDecoder().decode(AppSettings.self, from: data)
            guard decoded.schemaVersion == AppSettings.currentSchemaVersion else {
                // V1.0 只有 schema 1；未来版本在此迁移。当前按损坏处理。
                backupCorrupt(data)
                return (.default, true)
            }
            return (SettingsValidation.clampedForLoad(decoded), false)
        } catch {
            backupCorrupt(data)
            return (.default, true)
        }
    }

    public func save(_ settings: AppSettings) throws {
        let data = try JSONEncoder().encode(settings)
        defaults.set(data, forKey: Self.key)
    }

    private func backupCorrupt(_ data: Data) {
        let stamp = ISO8601DateFormatter().string(from: Date())
        defaults.set(data, forKey: Self.corruptBackupKeyPrefix + stamp)
    }
}
```

### 6.4 设置服务与广播（`Application/SettingsService.swift`）

```swift
import Foundation

/// Application 层。不依赖 SwiftUI/AppKit/Combine。
/// 持有当前设置；所有写入先校验，无效值拒绝且不落盘；有效变更按分类广播。
@MainActor
public final class SettingsService {
    public private(set) var current: AppSettings
    /// 启动加载时若发生损坏回退，保存一次性提示文案，由 UI 层展示一次后清除。
    public private(set) var recoveryNotice: String?

    private let store: any SettingsStoring
    private var continuations: [UUID: AsyncStream<AppSettingsChange>.Continuation] = [:]

    public init(store: any SettingsStoring) {
        self.store = store
        let (loaded, recovered) = store.load()
        self.current = loaded
        self.recoveryNotice = recovered
            ? "设置文件已损坏，已回退到默认值。原文件已备份在偏好设置中。" : nil
    }

    /// 订阅设置变更。订阅后立即收到一次以 current 为内容、包含全部分类的初始事件。
    public func changes() -> AsyncStream<AppSettingsChange> {
        let id = UUID()
        let current = self.current
        return AsyncStream { continuation in
            continuations[id] = continuation
            continuation.yield(AppSettingsChange(
                previous: current, current: current,
                changedCategories: Set(SettingsCategory.allCases)))
            continuation.onTermination = { [weak self] _ in
                Task { @MainActor [weak self] in self?.continuations[id] = nil }
            }
        }
    }

    /// 校验并保存。无效值：返回 false，携带 issues，不落盘、不广播。
    @discardableResult
    public func apply(_ newValue: AppSettings) -> Result<Void, [SettingsIssue]> {
        let issues = SettingsValidation.validate(newValue)
        guard issues.isEmpty else { return .failure(issues) }
        let old = current
        do { try store.save(newValue) } catch {
            return .failure([.init(field: .logDirectory,
                                   message: "设置保存失败：\(error.localizedDescription)")])
        }
        current = newValue
        let change = AppSettingsChange(previous: old, current: newValue,
                                       changedCategories: Self.diff(old, newValue))
        continuations.values.forEach { $0.yield(change) }
        return .success(())
    }

    public func resetToDefaults() {
        _ = apply(.default)
    }

    public func clearRecoveryNotice() { recoveryNotice = nil }

    private static func diff(_ a: AppSettings, _ b: AppSettings) -> Set<SettingsCategory> {
        var set = Set<SettingsCategory>()
        if a.serialDefaults != b.serialDefaults { set.insert(.serialDefaults) }
        if a.display != b.display { set.insert(.display) }
        if a.dataFormat != b.dataFormat { set.insert(.dataFormat) }
        if a.retention != b.retention { set.insert(.retention) }
        if a.logging != b.logging { set.insert(.logging) }
        if a.sessionAndPresets != b.sessionAndPresets { set.insert(.sessionAndPresets) }
        return set
    }
}
```

### 6.5 设置广播接线（`Features/Session/SessionSettingsBridge.swift`）

生效方式严格按需求文档 §7.1 逐行落实：

```swift
import Foundation

/// SessionViewModel 的扩展（04 文档定义的类型的扩展，不修改 04 文件本体）。
/// 把设置变更投影到运行中的会话。铁律：
/// - 不丢当前记录、不重写已有原始字节；
/// - 不静默修改当前连接参数；
/// - 进行中的日志保持启动时策略快照。
extension SessionViewModel {
    public func observeSettings(_ service: SettingsService) {
        let stream = service.changes()
        settingsObservationTask?.cancel()
        settingsObservationTask = Task { @MainActor [weak self] in
            for await change in stream {
                guard let self, !Task.isCancelled else { return }
                self.applySettingsChange(change)
            }
        }
    }

    private func applySettingsChange(_ change: AppSettingsChange) {
        for category in change.changedCategories {
            switch category {
            case .display:
                // 即时生效：只重建展示投影，RecordStore 原始数据不动 → 不丢记录。
                rebuildConsoleProjection()
            case .dataFormat:
                // 下一次显示或发送生效：既有记录的 displayFormat 快照不变，
                // 仅更新"后续记录/后续发送"使用的编码参数。
                updateActiveEncodingPreferences(change.current.dataFormat)
            case .retention:
                // 即时生效：下调立即按最早优先裁剪（03 文档 RecordStore 淘汰逻辑）。
                let r = change.current.retention
                Task { await recordStore.applyRetentionLimits(
                    maxCount: r.maxRecordCount, maxBytes: r.maxBufferBytes) }
            case .logging:
                // 仅新日志文件生效：06 的 startLogging() 每次读取 SettingsService.current，
                // 此处对运行中的日志不做任何推送。
                break
            case .serialDefaults:
                // 新会话或下一次连接生效：当前连接参数不被静默修改，此处无操作。
                break
            case .sessionAndPresets:
                // 重启后恢复：启动流程读取，运行中无操作。
                break
            }
        }
    }
}
```

```swift
import Observation

@MainActor @Observable
public final class SettingsViewModel {
    /// 草稿：控件直接绑定；每次编辑经校验后即时保存（设计规范 §8.4）。
    public private(set) var draft: AppSettings
    /// 字段级校验错误，就近展示在对应控件下方；不使用 Alert。
    public private(set) var fieldErrors: [SettingsField: String] = [:]
    public var showResetConfirmation = false

    public init(service: SettingsService) {
        self.service = service
        self.draft = service.current
        self.recoveryNotice = service.recoveryNotice
    }

    /// 通用编辑入口：修改草稿 → 校验 → 有效则立即保存；无效则草稿回滚该字段并显示就近错误。
    public func update<T: Equatable>(_ keyPath: WritableKeyPath<AppSettings, T>, value: T) {
        let oldValue = draft[keyPath: keyPath]
        draft[keyPath: keyPath] = value
        switch service.apply(draft) {
        case .success:
            fieldErrors = [:]
        case .failure(let issues):
            draft[keyPath: keyPath] = oldValue   // 无效值不保存，UI 回到上一个有效值
            fieldErrors = Dictionary(issues.map { ($0.field, $0.message) },
                                     uniquingKeysWith: { a, _ in a })
        }
    }

    public func requestResetToDefaults() { showResetConfirmation = true }

    public func confirmResetToDefaults() {
        service.resetToDefaults()
        draft = service.current
        fieldErrors = [:]
        showResetConfirmation = false
    }

    public func dismissRecoveryNotice() {
        service.clearRecoveryNotice()
        recoveryNotice = nil
    }
}
```

**预设**（`CommandMenu("预设")`）：

**视图**（插入系统"视图"菜单，`CommandGroup(after: .sidebarCommands)`）：
