- 禁止为通过性能测试而降低 RecordStore 上限默认值（50,000 条 / 16 MiB）或篡改测试数据源速率。

- 禁止只在 Xcode Debug 运行验证签名/沙盒——必须以 Release 签名构建、从 Finder/命令行启动验证。
- 禁止向应用包（.app Bundle）内写入日志或任何用户数据。
- 禁止引入 00/本文档未登记的第三方依赖（包括 fastlane 等发布工具链；发布脚本一律用 xcodebuild / codesign / xcrun notarytool 系统工具）。
- 禁止修改 01–07 已定稿的接口签名；确有障碍时按防漫游规则记录并停止。

本章给出可直接照抄的实现契约。P3 以配置/脚本/检查单为主，Swift 接口集中在批处理、退出确认、错误呈现与可访问性四块。

### 6.1 UI 刷新批处理器

```swift
import Foundation

/// UI 刷新批处理配置（00 文档 §3.3：高流量下批处理到 16–50 ms）
struct ConsoleBatchConfig: Sendable, Equatable {
    /// 批处理窗口（毫秒），合法区间 16...50，默认 33
    var windowMilliseconds: Int
    /// 单批最多合并的记录条数，超过则提前 flush（防单批过大掉帧）
    var maxRecordsPerBatch: Int

    static let `default` = ConsoleBatchConfig(windowMilliseconds: 33, maxRecordsPerBatch: 500)

    init(windowMilliseconds: Int = 33, maxRecordsPerBatch: Int = 500) {
        self.windowMilliseconds = min(max(windowMilliseconds, 16), 50)
        self.maxRecordsPerBatch = max(maxRecordsPerBatch, 1)
    }
}

/// RX/TX 记录进入 UI 前的批处理器。
/// 低流量：记录到达后至多等待一个窗口即刷新（保证延迟 ≤ window + 一帧）。
/// 高流量：窗口内到达的记录合并为一批，MainActor 每窗口最多刷新一次。
@MainActor
final class ConsoleBatchBuffer: ObservableObject {
    /// 已合并、等待渲染的记录（按到达时间升序）
    @Published private(set) var pending: [CommunicationRecord] = []
    /// 截断统计（RecordStore 淘汰时由 SessionViewModel 更新）
    @Published private(set) var truncationCount: Int = 0
    @Published private(set) var lastTruncationAt: Date?

    private let config: ConsoleBatchConfig
    private var flushTask: Task<Void, Never>?
    /// 批次落盘回调：由 SessionViewModel 注入，把一批记录应用到控制台投影
    var onFlush: ([CommunicationRecord]) -> Void = { _ in }

    init(config: ConsoleBatchConfig = .default) {
        self.config = config
    }

    /// RecordStore 产出新记录时调用（必须已在 MainActor）
    func enqueue(_ record: CommunicationRecord) {
        pending.append(record)
        if pending.count >= config.maxRecordsPerBatch {
            flush()
        } else {
            scheduleFlushIfNeeded()
        }
    }

    func reportTruncation(count: Int, at: Date) {
        truncationCount = count
        lastTruncationAt = at
    }

    /// 暂停显示时停止 flush（后台入缓存与写日志不受影响，由 RecordStore 保证）
    func setPaused(_ paused: Bool) {
        if paused {
            flushTask?.cancel(); flushTask = nil
        } else if !pending.isEmpty {
            scheduleFlushIfNeeded()
        }
    }

    /// 立即清空待渲染记录（清空控制台时调用；不影响 RecordStore 与日志）
    func clear() {
        flushTask?.cancel(); flushTask = nil
        pending.removeAll()
    }

    func flush() {
        flushTask?.cancel(); flushTask = nil
        guard !pending.isEmpty else { return }
        let batch = pending
        pending.removeAll(keepingCapacity: true)
        onFlush(batch)
    }

    private func scheduleFlushIfNeeded() {
        guard flushTask == nil else { return }
        let window = config.windowMilliseconds
        flushTask = Task { [weak self] in
            try? await Task.sleep(for: .milliseconds(window))
            guard !Task.isCancelled else { return }
            self?.flush()
        }
    }
}
```

### 6.2 退出确认

```swift
import AppKit

/// 退出前需要用户确认的脏状态快照（由 SessionViewModel 聚合提供）
struct ExitDirtyState: Equatable, Sendable {
    var isLoggingActive: Bool
    var isAutoSendRunning: Bool
    var isPresetLoopRunning: Bool
    var hasUnsavedPresetEdits: Bool
    var isConnected: Bool

    var requiresConfirmation: Bool {
        isLoggingActive || hasUnsavedPresetEdits || isAutoSendRunning
            || isPresetLoopRunning || isConnected
    }
}

/// 退出确认结果
enum ExitDecision {
    case stopAndQuit   // 停止日志/发送/会话，flush 完成后退出
    case cancel        // 取消退出，一切照旧
}

/// 应用退出拦截器：实现 NSApplicationDelegate 的 applicationShouldTerminate。
/// 确认退出后先执行 cleanup（停止调度、关日志 flush、关端口），全部完成才 reply(.terminateNow)。
@MainActor
final class ExitConfirmationController: NSObject, NSApplicationDelegate {
    /// 由 App 注入：返回当前脏状态
    var dirtyStateProvider: () -> ExitDirtyState = { ExitConfirmationController.clean }
    /// 由 App 注入：停止并退出前的清理（停止自动发送/预设循环 → flush 并关闭日志 → 关闭端口）
    var cleanup: () async -> Void = {}
    /// 由 App 注入：弹出确认对话框，返回用户选择（默认实现见 7.4.2）
    var confirm: (ExitDirtyState) async -> ExitDecision = { _ in .stopAndQuit }

    static let clean = ExitDirtyState(isLoggingActive: false, isAutoSendRunning: false,
                                      isPresetLoopRunning: false, hasUnsavedPresetEdits: false,
                                      isConnected: false)

    func applicationShouldTerminate(_ sender: NSApplication) -> NSApplication.TerminateReply {
        let state = dirtyStateProvider()
        guard state.requiresConfirmation else { return .terminateNow }
        Task { @MainActor in
            let decision = await confirm(state)
            switch decision {
            case .cancel:
                sender.reply(toApplicationShouldTerminate: false)
            case .stopAndQuit:
                await cleanup()
                sender.reply(toApplicationShouldTerminate: true)
            }
        }
        return .terminateLater
    }
}
```

### 6.3 错误呈现（三段式 + 可复制详情）

```swift
import Foundation

/// 用户可见错误的三段式文案（需求文档 §13：发生了什么、影响是什么、可以怎么做）
struct PresentedError: Equatable, Sendable, Identifiable {
    let id = UUID()
    /// 发生了什么：一句话，含对象名（如"无法打开 /dev/cu.usbserial"）
    var what: String
    /// 影响是什么：如"当前未连接，无法收发数据；正在记录的日志已停止"
    var impact: String
    /// 可以怎么做：可行动建议，如"检查设备是否被其他应用占用，然后点重试或刷新端口列表"
    var suggestion: String
    /// 可复制详情：分类、系统错误码、时间、原始描述（Detail 展开区展示）
    var technicalDetail: String
    /// 是否可重试（决定弹窗是否出现"重试"按钮）
    var retryable: Bool
}

extension PresentedError {
    /// 组装可复制详情文本（供展开区"拷贝"按钮写入剪贴板）
    var copyableReport: String {
        """
        \(what)
        \(impact)
        \(suggestion)
        ---
        \(technicalDetail)
        """
    }
}
```

### 6.5 Liquid Glass 可用性封装

enum VisualEffectsSupport {

### 7.1 性能工程

#### 7.1.1 伪终端高压数据源脚本

- 做什么：编写 `Scripts/pty_stress.py`。使用 `pty.openpty()` 创建伪终端对；以 `os.write` 向 master 持续写入循环递增的带序号数据块（如 `#00001234:55 AA ... \r\n`，每块 64 字节）；通过控制每次写入间隔把有效速率稳定在参数指定波特率（默认 115200 bps ≈ 11520 字节/秒）；支持 `--baud`、`--duration-min`（默认 30）、`--bidirectional`（同时读回显校验）；结束时打印总字节数、平均速率、写入次数。退出时清理 fd。
- 产出：`Scripts/pty_stress.py`，`chmod +x`。
- 自验证：运行 `python3 Scripts/pty_stress.py --baud 115200 --duration-min 1`，用 `cat <打印的slave路径>` 或应用接收 1 分钟，核对应用 RX 计数 × 平均块大小 ≈ 11520×60（±5%）。

#### 7.1.3 R5 控制台长文本性能评估与决策

- 做什么：对控制台做量化评估，决策保留 `ScrollView + LazyVStack` 还是迁移 NSTextView 桥接。
- 评估方法（在 7.1.1 数据源 115200 bps 输入下，RecordStore 上限 50,000 条场景）：
  1. Time Profiler 采样 60 秒，统计 MainActor 上控制台渲染相关占用；
  2. 记录满 50,000 条后：向上滚动到顶部再回底部的帧率（用 Instruments Core Animation FPS 或 Xcode FPS gauge）；
  3. 文本选择/拷贝大批量记录的响应时间；
  4. Allocations 记录控制台投影相关常驻内存。

  - MainActor 渲染占用 > 30% 或出现肉眼可见滚动掉帧（FPS < 50）；

## 10. 防漫游约束

重申 00 文档第 6 节全部规则，并对本文档特别强调：

2. **不引入未列依赖**：发布工具只用 xcodebuild/codesign/notarytool/stapler/hdiutil/ditto；不引入 fastlane、Sparkle 或任何新 SPM 包。
3. **R5 决策必须依据 7.1.3 的量化数据**，不得凭感觉直接迁移或直接保留；结论必须回写 00 文档第 9 节（这是本文档唯一允许修改 00 文档的位置）。
4. 需求文档、设计规范、Figma 原型冲突时按 **需求文档 > 设计规范 > Figma 原型** 取舍，冲突点记录到 00 文档第 9 节并停止该分支，不自行发挥。
5. 不实现任何 Out of scope 项（自动更新、多会话、协议解析、导出可读文本、遥测等），即使是"顺手"。
6. 每个实现步骤的自验证必须通过后再进入下一步；Instruments 与实机矩阵不得以"应该没问题"跳过。
7. 签名与发布步骤出现凭据/证书缺失时，停止并请示，不得自签 ad-hoc 构建冒充 RC。
