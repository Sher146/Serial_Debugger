## 6. 接口与数据模型

### 6.1 设计令牌落地

**冲突记录（以设计规范优先）**：原型 `index.css` 硬编码深色值且 accent 固定 `#2997ff`；设计规范 §2/§8 要求"依赖系统 accent color、semantic colors 与深浅色"。决策：**accent 一律用 `Color.accentColor`（系统强调色），不建 accent 资产**；RX/TX/警告/错误四个语义色建资产双变体——深色取原型基准值，浅色按系统语义色调校以保证白底对比度。此决策优先级依据 00 §1（需求文档 > 设计规范 > Figma 原型），已与设计规范一致，无遗留冲突。

### 6.2 引用契约（来自 02/03，照抄其定义，此处仅锁定 UI 消费面）

```swift
// —— 02 Domain（示意签名，以 02 文档实际定义为准）——
struct SerialPortDescriptor: Sendable, Equatable, Identifiable {
    var id: String { path }
    let path: String                 // /dev/cu.usbserial-0001
    let friendlyName: String?        // USB 厂商/产品可读名，可空
}
struct SerialConfiguration: Sendable, Equatable, Codable {
    var path: String; var baudRate: Int
    var dataBits: DataBits; var parity: Parity
    var stopBits: StopBits; var flowControl: FlowControl
}
enum SessionState: Sendable, Equatable {
    case disconnected
    case opening(port: SerialPortDescriptor)
    case connected(port: SerialPortDescriptor)
    case closing
    case error(SerialPortError)
}
struct CommunicationRecord: Sendable, Equatable, Identifiable {
    let id: UUID; let timestamp: Date
    let direction: RecordDirection   // .rx / .tx
    let rawBytes: Data               // 唯一事实来源；展示字符串一律由此派生
    let source: SendSource           // manual / autoSend / preset
    // displayFormat、sendResult、presetSnapshot 详见 02
}
enum DisplayFormat: Sendable { case ascii, hex }
enum LineEnding: Sendable, CaseIterable { case none, cr, lf, crlf }
enum PayloadValidator {                       // 02 提供
    static func validate(_ text: String, mode: DisplayFormat) -> Result<Data, ValidationError>
}

// —— 03 Application（示意签名，以 03 文档实际定义为准）——
enum LogStatus: Sendable, Equatable { case inactive; case recording(fileName: String) }
enum SessionEvent: Sendable {
    case stateChanged(SessionState)
    case portsUpdated([SerialPortDescriptor])
    case recordsAppended([CommunicationRecord]) // 已由协调器按 16–50ms 批处理
    case recordsCleared
    case countersUpdated(rxBytes: UInt64, txBytes: UInt64, truncationCount: Int)
    case logStatusChanged(LogStatus)
}
protocol SessionCoordinating: AnyObject, Sendable {
    func eventStream() -> AsyncStream<SessionEvent>
    func refreshPorts() async
    func connect(_ configuration: SerialConfiguration) async
    func cancelOpening() async
    func disconnect() async
    func sendManual(payload: Data, mode: DisplayFormat, lineEnding: LineEnding) async throws
    func clearRecords() async
}
```

### 6.3 `SessionViewModel`（完整定义，照抄实现）

```swift
// Features/Session/SessionViewModel.swift
import Foundation
import Observation
import SwiftUI

/// HEX 输入错误（UI 投影；判定由 Domain PayloadValidator 完成，此处仅区分提示文案）。
enum HexInputIssue: Sendable, Equatable {
    case oddLength          // 去除空白后字符数为奇数
    case illegalCharacter   // 含 0-9 A-F 以外字符
    case empty              // 无有效字节
    var message: String {
        switch self {
        case .oddLength:         return String(localized: "字节数为奇数，HEX 字符串长度须为偶数")
        case .illegalCharacter:  return String(localized: "包含非法字符，仅允许 0-9 A-F")
        case .empty:             return String(localized: "发送内容为空")
        }
    }
}

/// 状态徽标四态（三重表达：圆点颜色 + 文字 + 图标性发光/呼吸）。
enum ConnectionBadge: Equatable {
    case disconnected, connecting, connected, error
    var text: String {
        switch self {
        case .disconnected: return String(localized: "未连接")
        case .connecting:   return String(localized: "连接中…")
        case .connected:    return String(localized: "已连接")
        case .error:        return String(localized: "连接错误")
        }
    }
    var dotColor: Color {
        switch self {
        case .disconnected: return DesignTokens.Colors.textSecondary
        case .connecting:   return DesignTokens.Colors.warning
        case .connected:    return DesignTokens.Colors.tx
        case .error:        return DesignTokens.Colors.error
        }
    }
}

@MainActor @Observable
final class SessionViewModel {
    // MARK: 连接状态（投影，只读）
    private(set) var sessionState: SessionState = .disconnected
    private(set) var availablePorts: [SerialPortDescriptor] = []
    private(set) var lastError: SerialPortError?

    // MARK: 连接参数（用户意图，双向绑定）
    var selectedPortPath: String?
    var baudRate: Int = 115200
    var dataBits: DataBits = .eight
    var parity: Parity = .none
    var stopBits: StopBits = .one
    var flowControl: FlowControl = .none

    // MARK: 控制台
    private var records: [CommunicationRecord] = []     // 唯一存储；暂停时仍追加
    private var frozenCount: Int?                       // 非 nil 表示暂停中，视图只渲染 prefix
    private(set) var pausedPendingCount: Int = 0
    var displayFormat: DisplayFormat = .ascii { didSet { displayCache.removeAll() } }
    var showsTimestamps = true
    var followsTail = true                              // “跟随”开关；上滚解除见 ReceiveConsoleView
    private var displayCache: [UUID: String] = [:]      // 按当前 displayFormat 惰性缓存

    // MARK: 发送
    var sendText = "" { didSet { revalidateSend() } }
    var sendMode: DisplayFormat = .ascii { didSet { revalidateSend() } }
    var lineEnding: LineEnding = .crlf
    private(set) var hexIssue: HexInputIssue?

    // MARK: 计数 / 日志 / 截断
    private(set) var rxBytes: UInt64 = 0
    private(set) var txBytes: UInt64 = 0
    private(set) var truncationCount = 0
    private(set) var logStatus: LogStatus = .inactive

    // MARK: 布局
    var isPresetInspectorVisible = true

    static let baudRates = [300, 1200, 2400, 4800, 9600, 19200,
                            38400, 57600, 115200, 230400, 460800, 921600] // 12 档

    private let coordinator: any SessionCoordinating
    private var eventTask: Task<Void, Never>?

    init(coordinator: any SessionCoordinating) {
        self.coordinator = coordinator
        subscribe()
    }
    isolated deinit { eventTask?.cancel() }

    // MARK: 派生状态
    var badge: ConnectionBadge {
        switch sessionState {
        case .disconnected: return .disconnected
        case .opening:      return .connecting
        case .connected:    return .connected
        case .closing:      return .connected   // closing 徽标维持绿点，按钮置“断开中…”
        case .error:        return .error
        }
    }
    /// 连接中/已连接/关闭中：锁定影响打开过程的参数（端口、波特率、高级参数）
    var isParameterLocked: Bool {
        switch sessionState { case .disconnected, .error: return false; default: return true }
    }
    var isConnected: Bool { if case .connected = sessionState { return true } else { return false } }
    /// 选中端口不在当前列表 → 灰显“上次使用的端口不可用”
    var isSelectedPortMissing: Bool {
        guard let p = selectedPortPath else { return false }
        return !availablePorts.contains { $0.path == p }
    }
    /// 高级参数 chip 文案，如 “115200 · 8N1”
    var paramsChipLabel: String {
        let p = switch parity { case .none: "N"; case .even: "E"; case .odd: "O" }
        let s = switch stopBits { case .one: "1"; case .two: "2" }
        let d = switch dataBits { case .eight: "8"; case .seven: "7" }
        return "\(baudRate) · \(d)\(p)\(s)"
    }
    var canSend: Bool { isConnected && hexIssue == nil && !sendText.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty }
    var visibleRecords: [CommunicationRecord] {
        if let frozenCount { return Array(records.prefix(frozenCount)) }
        return records
    }
    var recordCount: Int { visibleRecords.count }

    // MARK: 事件订阅（单向数据流唯一入口）
    private func subscribe() {
        eventTask = Task { [weak self] in
            guard let self else { return }
            for await event in coordinator.eventStream() {
                guard !Task.isCancelled else { return }
                apply(event)
            }
        }
    }

    private func apply(_ event: SessionEvent) {
        switch event {
        case .stateChanged(let s):
            sessionState = s
            if case .connected = s { rxBytes = 0; txBytes = 0 }   // 新会话计数归零（UI 层决策，见 §6.7）
            if case .error(let e) = s { lastError = e }
        case .portsUpdated(let ports):
            availablePorts = ports
            if selectedPortPath == nil { selectedPortPath = ports.first?.path }
        case .recordsAppended(let batch):
            records.append(contentsOf: batch)
            if frozenCount != nil { pausedPendingCount += batch.count }
        case .recordsCleared:
            records.removeAll(); displayCache.removeAll()
            frozenCount = nil; pausedPendingCount = 0
        case .countersUpdated(let rx, let tx, let trunc):
            rxBytes = rx; txBytes = tx; truncationCount = trunc
        case .logStatusChanged(let s):
            logStatus = s
        }
    }

    // MARK: 用户意图
    func refreshPorts() { Task { await coordinator.refreshPorts() } }

    func connect() {
        guard let path = selectedPortPath, !isSelectedPortMissing else { return }
        let config = SerialConfiguration(path: path, baudRate: baudRate,
                                         dataBits: dataBits, parity: parity,
                                         stopBits: stopBits, flowControl: flowControl)
        Task { await coordinator.connect(config) }
    }
    /// 连接按钮统一入口：opening → 取消；connected → 断开；其余 → 连接
    func primaryConnectAction() {
        switch sessionState {
        case .opening:    Task { await coordinator.cancelOpening() }
        case .connected:  Task { await coordinator.disconnect() }
        case .closing:    break                                    // 按钮此时禁用
        default:          connect()
        }
    }

    func togglePause() {
        if frozenCount == nil { frozenCount = records.count; pausedPendingCount = 0 }
        else { frozenCount = nil; pausedPendingCount = 0 }
    }
    var isPaused: Bool { frozenCount != nil }

    func clearConsole() { Task { await coordinator.clearRecords() } } // 仅内存显示；不影响日志与字节计数

    func send() {
        guard canSend, case .success(let payload) = PayloadValidator.validate(sendText, mode: sendMode) else { return }
        let text = sendText
        Task {
            try? await coordinator.sendManual(payload: payload, mode: sendMode, lineEnding: lineEnding)
            // 成功排入队列即清空输入；失败保留文本（错误经事件流进入 lastError / TX failed 记录）
            if (try? PayloadValidator.validate(self.sendText, mode: self.sendMode)) != nil,
               self.sendText == text { self.sendText = "" }
        }
    }

    func copyRecordToPasteboard(_ record: CommunicationRecord) {
        NSPasteboard.general.clearContents()
        NSPasteboard.general.setString(displayText(for: record), forType: .string)
    }

    /// 展示字符串：HEX 一律基于 rawBytes（UTF-8 字节），禁止 charCode；ASCII 无法解码处以 U+FFFD 占位显示。
    func displayText(for record: CommunicationRecord) -> String {
        if let cached = displayCache[record.id] { return cached }
        let text: String
        switch displayFormat {
        case .hex:
            text = record.rawBytes.map { String(format: "%02X", $0) }.joined(separator: " ")
        case .ascii:
            text = String(decoding: record.rawBytes, as: UTF8.self)
        }
        displayCache[record.id] = text
        return text
    }

    private func revalidateSend() {
        guard sendMode == .hex else { hexIssue = nil; return }
        let cleaned = sendText.filter { !$0.isWhitespace }
        if cleaned.isEmpty { hexIssue = nil; return }                 // 空输入不报错，仅禁用发送
        if cleaned.count % 2 != 0 { hexIssue = .oddLength; return }
        if cleaned.contains(where: { !$0.isHexDigit }) { hexIssue = .illegalCharacter; return }
        hexIssue = nil
    }
}
```

> `send()` 中"成功清空、失败保留"的判定以 03 的 `sendManual` 语义为准：若其实现为"排入队列即返回、写失败异步入记录"，则直接无条件清空输入，写失败通过 TX 记录 `failed` 标记与 `lastError` 呈现（需求 §5.3）。两种实现对本文档视图层透明，按 03 实际契约二选一并在代码注释标注。

### 6.5 `ConsoleFormatting`

```swift
// Features/Session/ConsoleFormatting.swift
import Foundation

enum ConsoleFormatting {
    /// HH:MM:SS.mmm（24 小时制 + 3 位毫秒）；DateFormatter 昂贵，静态复用
    static let timestamp: DateFormatter = {
        let f = DateFormatter(); f.dateFormat = "HH:mm:ss.SSS"; f.locale = Locale(identifier: "en_US_POSIX")
        return f
    }()
    /// B / KB / MB，一位小数
    static func byteCount(_ n: UInt64) -> String {
        switch n {
        case ..<1024: return "\(n) B"
        case ..<1_048_576: return String(format: "%.1f KB", Double(n) / 1024)
        default: return String(format: "%.1f MB", Double(n) / 1_048_576)
        }
    }
}
```

### 6.6 呼吸点（原型 pulse 动画的 SwiftUI 实现）

```swift
/// 原型 CSS `animation: pulse 1s infinite` 未定义，禁止照抄；统一用本组件。
struct BreathingDot: View {
    let color: Color
    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    var body: some View {
        Circle().fill(color)
            .frame(width: DesignTokens.Metrics.statusDotSize,
                   height: DesignTokens.Metrics.statusDotSize)
            .phaseAnimator([0.35, 1.0]) { view, phase in
                view.opacity(reduceMotion ? 1.0 : phase)   // Reduce Motion：静态全亮
            } animation: { _ in .easeInOut(duration: 0.8) }
    }
}
```

用于 connecting 徽标（05 复用于"自动发送运行中"点）。connected 发光用 `.shadow(color:radius: 3)`，非动画。
