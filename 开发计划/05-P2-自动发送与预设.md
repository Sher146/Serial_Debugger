### Out of scope（明确不做）

- 菜单命令与快捷键注册（Cmd+Shift+Return 等）：属 07 文档；本文档只在 `SessionViewModel` 暴露可供菜单调用的方法。

- 发送编辑器本体、控制台、工具栏、状态栏组件本体：属 04 文档；本文档只做最小接线修改。

- 多会话、协议解析、脚本、导出可读文本（见 00 文档 §2 V1.0 不做清单）。

### 明确禁止事项

- 禁止照搬 Figma 原型瑕疵（见 4.3）：循环按数组顺序 round-robin、自动发送使用独立"上次手动发送文本"而非当前编辑器内容、循环间隔下限 100 ms。
- 禁止在 Presentation 层直接读写文件、解析 HEX、创建 Timer/ Task 调度。
- 禁止使用 `Timer` / `DispatchQueue.asyncAfter` 做发送调度（无法与 actor 串行化及取消语义正确组合），统一用 `ContinuousClock` + 结构化并发。
- 禁止引入新第三方依赖（本文档零新增依赖）。
- 禁止自动发送在应用启动时自动恢复运行（需求 §7.1 硬性规定）。

```swift
// Domain/Scheduling/ScheduledPayload.swift
import Foundation

/// 一次调度发送的完整载荷：字节在调度器外（ViewModel/Service）完成编码后传入。
public struct ScheduledPayload: Sendable {
    public let bytes: Data
    public let source: SendSource       // .autoSend 或 .preset
    public let preset: PresetSnapshot?  // source == .preset 时非 nil

    public init(bytes: Data, source: SendSource, preset: PresetSnapshot? = nil) {
        self.bytes = bytes
        self.source = source
        self.preset = preset
    }
}

/// 调度器运行模式（互斥：同一时刻至多一个非 idle）。
public enum SchedulerMode: Equatable, Sendable {
    case idle
    case autoSend(intervalMs: Int)
    case presetLoop(intervalMs: Int, presetCount: Int)
}

/// 调度停止原因；状态栏据此显示（需求 §12.2"状态栏显示停止原因"）。
public enum SchedulerStopReason: String, Equatable, Sendable {
    case userStopped            // 用户点击停止
    case supersededByAutoSend   // 被普通自动发送顶替
    case supersededByPresetLoop // 被预设循环顶替
    case payloadBecameInvalid   // 编辑器内容变为空/非法 HEX
    case noEnabledPresets       // 无已启用预设 / 运行中全部被停用
    case parameterChanged       // 编码模式/行结束符/端口参数变更
    case disconnected           // 端口关闭/设备拔出
    case sendFailed             // 底层写入失败（不可恢复）
    case appTerminating         // 应用退出流程
}
```

```swift
// Application/SendScheduler/SendScheduler.swift
import Foundation

/// 定时发送调度器（需求 §5.4、§12.2）。
///
/// 铁律实现方式：
/// - 防重入：单一循环 Task，每次 `send` 被 await 完成后才按单调时钟 sleep，
///   不存在并发 tick，UI 卡顿只会推迟不会堆积。
/// - 互斥：`startAutoSend` / `startPresetLoop` 开头先 `stopCurrent(reason: .supersededBy…)`。
/// - 取消：stop 置原因 → cancel Task → await 结束 → 发布 .idle 状态。
///
/// 由 SessionCoordinator 持有（00 文档 §3.3：协调器是唯一拥有调度任务的 actor）；
/// 本 actor 不直接触碰串口，发送经注入的 sendHandler 回到 SessionCoordinator 的 TX 队列。
public actor SendScheduler {

    public struct State: Equatable, Sendable {
        public var mode: SchedulerMode
        public var lastStopReason: SchedulerStopReason?
        public var sentCount: Int   // 本次运行已发送条数（启动时清零）
        public static let idle = State(mode: .idle, lastStopReason: nil, sentCount: 0)
    }

    public typealias SendHandler = @Sendable (ScheduledPayload) async throws -> Void
    /// 普通自动发送载荷提供者：每 tick 调用；返回 nil 表示内容已非法 → 停止(.payloadBecameInvalid)。
    public typealias AutoPayloadProvider = @Sendable () -> ScheduledPayload?
    /// 预设循环轮次提供者：每轮（round）开始调用一次，返回该轮快照，
    /// 已按 sortOrder 升序、仅含已启用项、字节已编码、preset 快照已生成；空数组 → 停止(.noEnabledPresets)。
    public typealias RoundProvider = @Sendable () async -> [ScheduledPayload]

    private let clock: SendClock
    private let sendHandler: SendHandler
    private var loopTask: Task<Void, Never>?
    private var state = State.idle
    private var continuations: [UUID: AsyncStream<State>.Continuation] = [:]

    public init(clock: SendClock, sendHandler: @escaping SendHandler) {
        self.clock = clock
        self.sendHandler = sendHandler
    }

    public var currentState: State { state }

    public func stateUpdates() -> AsyncStream<State> {
        let id = UUID()
        return AsyncStream { continuation in
            continuations[id] = continuation
            continuation.yield(state)
            continuation.onTermination = { [weak self] _ in
                Task { await self?.removeContinuation(id) }
            }
        }
    }

    /// 启动普通自动发送。调用方（ViewModel）已完成间隔校验与连接状态校验。
    public func startAutoSend(intervalMs: Int, payloadProvider: @escaping AutoPayloadProvider) async {
        await stopCurrent(reason: .supersededByAutoSend)
        state = State(mode: .autoSend(intervalMs: intervalMs), lastStopReason: state.lastStopReason, sentCount: 0)
        publish()
        loopTask = Task { [clock, sendHandler] in
            while !Task.isCancelled {
                guard let payload = payloadProvider() else { return } // 原因见下方 finish 判定
                do {
                    try await sendHandler(payload)          // 串行：await 完成才进入下一拍
                } catch is CancellationError {
                    return
                } catch {
                    return                                   // 发送失败：由协调器另行以 .sendFailed 停止
                }
                try? await clock.sleep(milliseconds: intervalMs) // 单调时钟重新计算，不堆积
            }
        }
    }

    /// 启动预设循环。单轮内顺序发送 roundProvider 返回的快照；每轮开始重新取最新已保存顺序（§12.2）。
    public func startPresetLoop(intervalMs: Int, roundProvider: @escaping RoundProvider) async {
        await stopCurrent(reason: .supersededByPresetLoop)
        state = State(mode: .presetLoop(intervalMs: intervalMs, presetCount: 0), lastStopReason: state.lastStopReason, sentCount: 0)
        publish()
        loopTask = Task { [clock, sendHandler] in
            while !Task.isCancelled {
                let round = await roundProvider()            // 每轮开始读取最新快照
                guard !round.isEmpty else { return }
                for payload in round {
                    if Task.isCancelled { return }
                    do {
                        try await sendHandler(payload)
                    } catch is CancellationError {
                        return
                    } catch {
                        return
                    }
                    try? await clock.sleep(milliseconds: intervalMs) // 条与条之间、轮与轮之间同间隔
                }
            }
        }
    }

    /// 停止并等待任务结束；幂等。发布 .idle 且携带停止原因。
    public func stop(reason: SchedulerStopReason) async {
        await stopCurrent(reason: reason)
    }

    // MARK: - Private

    private func stopCurrent(reason: SchedulerStopReason) async {
        guard let task = loopTask else {
            state.lastStopReason = reason
            return
        }
        loopTask = nil
        task.cancel()
        _ = await task.value                                  // 必须 await 旧任务结束（00 文档 §3.3）
        state.mode = .idle
        state.lastStopReason = reason
        publish()
    }

    private func publish() {
        for continuation in continuations.values { continuation.yield(state) }
    }

    private func removeContinuation(_ id: UUID) { continuations[id] = nil }
}
```

> 实现注：任务自然结束（provider 返回 nil / round 为空 / send 抛错）时，actor 外部由 SessionCoordinator 监听发送结果流，调用 `stop(reason:)` 收尾并发布原因；Task 内直接 return 只保证不再发送。测试中以虚拟时钟 + 受控 handler 验证该收尾路径。

```swift
// Application/Presets/PresetService.swift
import Foundation

/// 预设业务服务：CRUD、排序、启用集查询。不依赖 SwiftUI/AppKit。
public actor PresetService {

    public enum LoadOutcome: Equatable, Sendable {
        case loaded(count: Int)
        case corruptedRecovered(backupPath: String)  // 已备份损坏文件并回退空列表；UI 一次性提示
    }

    private let store: PresetFileStore
    private var presets: [Preset] = []
    private var continuations: [UUID: AsyncStream<[Preset]>.Continuation] = [:]

    public init(store: PresetFileStore) {
        self.store = store
    }

    /// 启动时调用一次；返回加载结果供 UI 决定是否显示"配置损坏已回退"提示（需求 §13）。
    @discardableResult
    public func load() async -> LoadOutcome {
        let outcome = await store.load()
        switch outcome {
        case .success(let loaded):
            presets = loaded.sortedForDisplay()
            return .loaded(count: loaded.count)
        case .recovered(let backupPath):
            presets = []
            return .corruptedRecovered(backupPath: backupPath.path)
        }
    }

    public func all() -> [Preset] { presets }

    /// 循环发送唯一取数口：仅已启用、严格按 sortOrder 升序（需求 §5.4）。
    public func enabledPresetsSorted() -> [Preset] {
        presets.filter(\.isEnabled).sortedForDisplay()
    }

    public func presetsUpdates() -> AsyncStream<[Preset]> { /* 同 SendScheduler 的 continuation 模式 */ }

    public func create(name: String, content: String, encoding: PayloadEncoding, note: String?) async throws -> Preset {
        let nextOrder = (presets.map(\.sortOrder).max() ?? -1) + 1
        let preset = Preset(name: name, content: content, encoding: encoding,
                            sortOrder: nextOrder, note: note?.nilIfBlank)
        presets.append(preset)
        try await persist()
        return preset
    }

    /// 显式保存语义：仅编辑 Sheet 点保存时调用；updatedAt 在此刷新。
    public func update(_ preset: Preset) async throws {
        guard let index = presets.firstIndex(where: { $0.id == preset.id }) else { return }
        var copy = preset
        copy.updatedAt = Date()
        presets[index] = copy
        try await persist()
    }

    public func delete(id: UUID) async throws {
        presets.removeAll { $0.id == id }
        try await persist()
    }

    public func setEnabled(_ isEnabled: Bool, id: UUID) async throws {
        guard let index = presets.firstIndex(where: { $0.id == id }) else { return }
        presets[index].isEnabled = isEnabled
        presets[index].updatedAt = Date()
        try await persist()
    }

    /// 拖动/键盘排序：把 id 移动到目标下标，重写全部 sortOrder 为 0…n-1 并持久化。
    public func move(id: UUID, toIndex destination: Int) async throws {
        guard let from = presets.firstIndex(where: { $0.id == id }) else { return }
        let item = presets.remove(at: from)
        let clamped = max(0, min(destination, presets.count))
        presets.insert(item, at: clamped)
        for (index, _) in presets.enumerated() { presets[index].sortOrder = index }
        try await persist()
    }

    private func persist() async throws {
        presets = presets.sortedForDisplay()
        try await store.save(presets)
        for continuation in continuations.values { continuation.yield(presets) }
    }
}

private extension Array where Element == Preset {
    func sortedForDisplay() -> [Preset] {
        sorted { ($0.sortOrder, $0.createdAt) < ($1.sortOrder, $1.createdAt) }
    }
}

private extension String {
    var nilIfBlank: String? { trimmingCharacters(in: .whitespacesAndNewlines).isEmpty ? nil : self }
}
```

```swift
// Infrastructure/Persistence/PresetFileStore.swift
import Foundation

/// presets.json 持久化：versioned JSON + 原子写入（临时文件 + replace，需求 §6.2）。
public actor PresetFileStore {

    public enum Outcome: Equatable {
        case success([Preset])
        case recovered(backupPath: URL)   // 损坏：已备份原文件并回退默认值
    }

    struct FileEnvelope: Codable {
        var schemaVersion: Int
        var presets: [Preset]
    }

    public static let currentSchemaVersion = 1

    private let fileURL: URL
    private let fileManager: FileManager

    /// 默认位置 ~/Library/Application Support/SerialMonitor/presets.json；测试注入临时目录。
    public init(fileURL: URL, fileManager: FileManager = .default) {
        self.fileURL = fileURL
        self.fileManager = fileManager
    }

    public func load() async -> Outcome {
        guard fileManager.fileExists(atPath: fileURL.path) else { return .success([]) }
        do {
            let data = try Data(contentsOf: fileURL)
            let envelope = try JSONDecoder.presetDecoder.decode(FileEnvelope.self, from: data)
            guard envelope.schemaVersion <= Self.currentSchemaVersion else {
                return await backupAndRecover(reason: "schemaVersion \(envelope.schemaVersion) 高于当前支持版本")
            }
            return .success(envelope.presets)
        } catch {
            return await backupAndRecover(reason: error.localizedDescription)
        }
    }

    /// 原子写入：先写同目录临时文件，再 replace 到目标路径；失败抛错由调用方提示，不留半截文件。
    public func save(_ presets: [Preset]) async throws {
        let directory = fileURL.deletingLastPathComponent()
        try fileManager.createDirectory(at: directory, withIntermediateDirectories: true)
        let envelope = FileEnvelope(schemaVersion: Self.currentSchemaVersion, presets: presets)
        let data = try JSONEncoder.presetEncoder.encode(envelope)
        let tempURL = directory.appendingPathComponent(".presets-\(UUID().uuidString).tmp")
        try data.write(to: tempURL, options: .atomic)
        _ = try fileManager.replaceItemAt(fileURL, withItemAt: tempURL)
    }

    /// 损坏回退（需求 §13）：备份为 presets.corrupted-<yyyyMMdd-HHmmss>.json，回退空列表；提示只发一次由 UI 控制。
    private func backupAndRecover(reason: String) async -> Outcome {
        let stamp = ISO8601DateFormatter.fileSafe.string(from: Date())
        let backupURL = fileURL.deletingPathExtension()
            .appendingPathExtension("corrupted-\(stamp).json")
        try? fileManager.moveItem(at: fileURL, to: backupURL)
        return .recovered(backupPath: backupURL)
    }
}

// Infrastructure/Persistence/SessionSnapshotStore.swift
import Foundation

/// 会话快照（需求 §6.2：与预设同目录、versioned JSON、原子写入）。
/// 本文档仅写入自动发送草稿；后续文档（07）扩展字段时必须升 schemaVersion 并写迁移。
public actor SessionSnapshotStore {

    struct Snapshot: Codable {
        var schemaVersion: Int
        var autoSendDraft: AutoSendDraft
    }

    private let fileURL: URL
    public init(fileURL: URL) { self.fileURL = fileURL }

    public func loadAutoSendDraft() async -> AutoSendDraft {
        guard let data = try? Data(contentsOf: fileURL),
              let snapshot = try? JSONDecoder().decode(Snapshot.self, from: data),
              snapshot.schemaVersion <= 1
        else { return AutoSendDraft() }                 // 损坏静默回退默认值（草稿非关键数据）
        return snapshot.autoSendDraft
    }

    public func saveAutoSendDraft(_ draft: AutoSendDraft) async throws {
        let snapshot = Snapshot(schemaVersion: 1, autoSendDraft: draft)
        let data = try JSONEncoder.presetEncoder.encode(snapshot)
        let tempURL = fileURL.deletingLastPathComponent()
            .appendingPathComponent(".session-\(UUID().uuidString).tmp")
        try FileManager.default.createDirectory(at: fileURL.deletingLastPathComponent(), withIntermediateDirectories: true)
        try data.write(to: tempURL, options: .atomic)
        _ = try FileManager.default.replaceItemAt(fileURL, withItemAt: tempURL)
    }
}
```

## 8. 验收标准

映射需求文档 §16：

- **AC-05 自动发送**——"正数间隔可稳定启动 / 停止；串口关闭或错误后立即停止。"细化：
  1. 间隔 10 / 1000 / 3,600,000 ms 均可启动；9 ms、0、负数、非数字、3,600,001 被拦截并就位提示。
  2. 运行中关闭端口 / 拔出设备（伪终端模拟）→ 1 个 tick 内停止，状态栏显示"自动发送已停止：连接断开"。
  3. 运行中切换 ASCII/HEX、行结束符或端口参数 → 先停止，原因 `.parameterChanged`。
  4. handler 人为延迟 2× 间隔 → 无并发发送、无追赶发送（发送次数 = 完成次数）。
  5. 启动预设循环 → 自动发送停止，状态栏显示"自动发送已停止：已启动预设循环"。
  6. 重启应用 → 间隔草稿恢复，但调度不自动运行（`schedulerState.mode == .idle`）。

- **AC-06 预设**——"增删改排、启用、单发、循环与重启恢复正确；日志可区分预设来源。"细化：
  1. 新建/编辑/删除/启用切换立即持久化；强杀重启后列表与顺序一致。
  2. 拖动排序与键盘排序后 `sortOrder` 连续（0…n-1）并持久化。
  3. 单发生成 `source == .preset` 且带 `PresetSnapshot` 的 TX 记录。
  4. 循环严格按 sortOrder 遍历已启用项；运行中停用某预设 → 下一轮起不再发送（本轮快照仍按 §12.2 完成）；全部停用 → 自动停止并提示。
  5. 无已启用项时"开始循环"不可启动并提示"没有已启用的预设"。
  6. 改名后发送：新记录用新名快照，旧记录保留旧名（日志可追溯，06 消费）。
  7. presets.json 损坏 → 备份文件生成、回退空列表、一次性提示且不阻塞启动。

- 两段式删除：第一次点击 → 按钮变红、tooltip/标签变"再次点击确认删除"，3 秒无操作自动还原；第二次点击执行删除。`PresetsViewModel` 持有 `deleteConfirmationID` 与 3 秒 Task（行移出屏幕/删除其他项时取消）。
- 拖动排序：`List` + `.onMove`；键盘替代：行 context menu 与快捷键 ⌥⌘↑ / ⌥⌘↓ 调用 `move(up:)`，并提供 `.accessibilityAction(named:)` "上移/下移"（可访问性硬性要求）。
- 循环控制区（底部固定）：间隔输入（ms，失焦/回车校验，非法就近红字提示，10–49 ms 显示橙色风险提示"间隔过小可能超出设备处理能力"）、"N 个已启用的预设将循环发送"动态文案、开始/停止按钮三态：可启动（绿）/ 运行中（红"停止循环"）/ 禁用（未连接或无已启用项，附原因 tooltip）。
- 显隐：工具栏检查器切换按钮（04 工具栏扩展点）；窗口宽度 < 1100 pt 自动收起（`GeometryReader` 或 `onGeometryChange`），状态写 UserDefaults；收窄收起不覆盖用户手动展开偏好，仅当次会话生效。

### 8.3 预设编辑 Sheet 交互规格（PresetEditSheet）

参照 `PresetEditModal.tsx`，落地为 SwiftUI `.sheet`（420 pt 宽，`Form` 风格）：

- 字段：名称（必填，自动聚焦）、内容（多行，等宽字体）、ASCII/HEX `Picker(.segmented)`、启用 `Toggle`、备注（可选单行）。
- 实时 HEX 校验：HEX 模式且内容非空时调用 `HexCodec.validate`；错误横幅就近显示在内容框下方（"HEX 字符串长度须为偶数" / "包含非法字符，仅允许 0-9 A-F"），不弹 alert、不修改用户输入。
- 保存语义：**显式保存**——Sheet 内编辑不落盘，仅点"保存"或 ⌘Enter（`.keyboardShortcut(.return, modifiers: .command)`）时 `PresetService.create/update`；Esc（`.keyboardShortcut(.escape)`）与"取消"走关闭流程。
- 保存按钮禁用：名称为空白 / 内容为空白 / HEX 校验失败，任一即禁用。
- 脏状态关闭确认：任一字段与初始值不同时关闭 → `confirmationDialog`"放弃未保存的修改？"（放弃修改 / 继续编辑）；未修改直接关闭。
- 编辑运行中循环涉及的预设：允许保存，按 §12.2 下一轮生效（快照语义），不中断循环。
